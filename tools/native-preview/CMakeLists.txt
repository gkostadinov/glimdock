cmake_minimum_required(VERSION 3.20)
project(homelab_native_preview LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED LVGL_SOURCE_DIR)
  message(FATAL_ERROR "Set LVGL_SOURCE_DIR to the LVGL 9.3.0 source directory")
endif()
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_BUILD_CONF_DIR "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl)
target_include_directories(lvgl PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../firmware/include")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../firmware/src/main.cpp")
  add_executable(native-preview renderer.cpp ../../firmware/src/main.cpp)
  target_include_directories(native-preview PRIVATE shims ../../firmware/src ../../firmware/include)
  target_include_directories(native-preview PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_definitions(native-preview PRIVATE HOMELAB_NATIVE_PREVIEW=1)
  target_link_libraries(native-preview PRIVATE lvgl)
  add_executable(native-gestures gestures.cpp ../../firmware/src/main.cpp)
  target_include_directories(native-gestures PRIVATE gesture-shims shims ../../firmware/src ../../firmware/include "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_definitions(native-gestures PRIVATE HOMELAB_NATIVE_PREVIEW=1)
  target_link_libraries(native-gestures PRIVATE lvgl)
endif()
