cmake_minimum_required(VERSION 3.20)
project(glimdock_web_emulator LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "Configure with emcmake: this target runs in WebAssembly")
endif()
if(NOT DEFINED LVGL_SOURCE_DIR OR NOT DEFINED ARDUINOJSON_SOURCE_DIR)
  message(FATAL_ERROR "Set LVGL_SOURCE_DIR and ARDUINOJSON_SOURCE_DIR")
endif()
set(HOMELAB_ROTATION 3 CACHE STRING "Display orientation")
if(NOT HOMELAB_ROTATION MATCHES "^[0-3]$")
  message(FATAL_ERROR "HOMELAB_ROTATION must be 0, 1, 2 or 3")
endif()
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_BUILD_CONF_DIR "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl)
target_include_directories(lvgl PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../firmware/include")
add_executable(firmware emulator.cpp ../../firmware/src/main.cpp)
target_include_directories(firmware PRIVATE shims ../native-preview/shims ../../firmware/src ../../firmware/include "${ARDUINOJSON_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(firmware PRIVATE HOMELAB_NATIVE_PREVIEW=1 GLIMDOCK_PUBLIC_FIRMWARE=1 HOMELAB_ROTATION=${HOMELAB_ROTATION} ARDUINOJSON_ENABLE_ARDUINO_STRING=0 ARDUINOJSON_ENABLE_ARDUINO_STREAM=0)
target_link_libraries(firmware PRIVATE lvgl)
target_compile_options(firmware PRIVATE -O2)
target_link_options(firmware PRIVATE -O2 "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sEXPORT_NAME=createGlimdockFirmware" "-sENVIRONMENT=web,node" "-sALLOW_MEMORY_GROWTH=1" "-sINITIAL_MEMORY=16777216" "-sSTACK_SIZE=262144" "-sFILESYSTEM=0" "-sNO_EXIT_RUNTIME=1" "-sDYNAMIC_EXECUTION=0" "-sEXPORTED_RUNTIME_METHODS=['ccall','UTF8ToString','HEAPU16']")
set_target_properties(firmware PROPERTIES SUFFIX ".js")
